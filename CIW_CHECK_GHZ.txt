procedure(busVal(prefix tsamp width)
  let((s i)
    s = 0
    for(i 0 width-1
      when(value(v(sprintf(nil "%s<%d>" prefix i) ?result "tran") tsamp) > 0.6
        s = s + 2**i))
    s))

foreach(item list(list("A" 0.9e-9 0) list("B" 1.9e-9 65534) list("C" 2.9e-9 21844))
  let((lbl tsamp expS sv cv)
    lbl = car(item)
    tsamp = cadr(item)
    expS = caddr(item)
    sv = busVal("/SOUT" tsamp 16)
    cv = value(v("/COUT" ?result "tran") tsamp)
    printf("Vector %s @ %g s: S=%d (0x%04x) expected %d  COUT=%g  %s\n"
      lbl tsamp sv sv expS cv
      if(sv == expS && cv > 0.6 then "PASS" else "FAIL"))))
